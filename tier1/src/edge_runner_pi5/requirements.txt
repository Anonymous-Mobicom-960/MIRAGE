# 🔴 THIS FILE IS DELIBERATELY NOT VERSION-PINNED, and that is a measured decision rather than an

# treat any version claim about this file as unverified unless it was measured on the Pi.

mediapipe==0.10.18   # PIN: last PyPI aarch64 (manylinux) wheel; needs Python <= 3.12 (no cp313 wheel).
                     # FaceMesh -> the 12 face scalars. If unavailable on the device, try 'mediapipe-rpi4'.
